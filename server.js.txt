const express = require('express');
const path = require('path');
const app = express();

const PORT = process.env.PORT || 3000;

// خدمة الملفات الثابتة في المجلد
app.use(express.static(__dirname));

// مسار لوحة التحكم
app.get('/dashboard', (req, res) => {
    res.sendFile(path.join(__dirname, 'tasmeemy_admin_dashboard_2.html'));
});

// المسار الرئيسي (صفحة الخدمة)
app.get('/', (req, res) => {
    res.sendFile(path.join(__dirname, 'tasmeemy_notebook_formatting_service_2.html'));
});

app.listen(PORT, () => {
    console.log(`Server is running on port ${PORT}`);
});
